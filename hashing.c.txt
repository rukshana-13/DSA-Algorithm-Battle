#include <stdio.h>

#define SIZE 101

struct Student {
    int id;
    char name[30];
    int used;
};

struct Student table[SIZE];

int collisions = 0;
int comparisons = 0;

int hashFunction(int id) {
    return id % SIZE;
}

void insert(int id, char name[]) {
    int index = hashFunction(id);
    int start = index;

    while (table[index].used) {
        if (table[index].id == id) {
            printf("Duplicate ID: %d\n", id);
            return;
        }

        collisions++;
        index = (index + 1) % SIZE;

        if (index == start) {
            printf("Hash table is full!\n");
            return;
        }
    }

    table[index].id = id;
    snprintf(table[index].name, 30, "%s", name);
    table[index].used = 1;
}

void search(int key) {
    int index = hashFunction(key);
    int start = index;

    comparisons = 0;

    while (table[index].used) {
        comparisons++;

        if (table[index].id == key) {
            printf("Student Found: %s\n", table[index].name);
            printf("Search Comparisons: %d\n", comparisons);
            return;
        }

        index = (index + 1) % SIZE;

        if (index == start)
            break;
    }

    printf("Student Not Found\n");
    printf("Search Comparisons: %d\n", comparisons);
}

void display() {
    printf("\nStudent Records:\n");

    for (int i = 0; i < SIZE; i++) {
        if (table[i].used)
            printf("%d - %s (Index %d)\n",
                   table[i].id, table[i].name, i);
    }
}

int main() {
    int n, id, key;
    char name[30];

    printf("Enter number of students: ");
    scanf("%d", &n);

    if (n < 1 || n > 50) {
        printf("Enter between 1 and 50 students.\n");
        return 0;
    }

    for (int i = 0; i < n; i++) {
        printf("Enter ID and Name: ");
        scanf("%d %29s", &id, name);
        insert(id, name);
    }

    display();

    printf("\nTotal Collisions: %d\n", collisions);

    printf("\nEnter ID to search: ");
    scanf("%d", &key);

    search(key);

    return 0;
}
